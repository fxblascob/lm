Pulsa aquí para detener el scroll:

En esta página generamos un scroll. A continuación podrás ver el código fuente de la misma: 

<HEAD>  
...   
    <SCRIPT LANGUAGE="JavaScript">    
	var texto="  Mira que bonito es mi scroll   ";    
	var longitud=texto.length;    
	var timeout;
	function scroll() {      
		texto=texto.substring(1,longitud-1)+texto.charAt(0);      
		window.status = texto;
     		timeout=setTimeout("scroll()",150);      
     	}
    	function finScroll() {
   		clearTimeout(timeout);
   		window.status="fin de scroll";
   		return true;
   	}
    </SCRIPT>
</HEAD>
<BODY onLoad="timeout=scroll();return true">

...

<form>
<input type="button" value=" detener " onClick="finScroll()">
</form>
