Nota: Si se introduce un día del mes superior a los estimados, por ejemplo 34 de diciembre, no dará error. Simplemente se irá pasando de mes hasta que el día esté dentro de los valores razonables. De esta manera el 34 de Diciembre será igual al 3 de febrero. El mismo caso pasa con los minutos, mientras que con los segundos lo que hace JavaScript es calcular el módulo de la cantidad introducida por el usuario así se garantiza que los segundos siempre estarán en un valor comprendido entre 0 y 59.

		function Dia_semana(dia){ 
			// dia es un entero que representa la día de la semana 
			// (0=domingo y 6=sábado)
			var aux="";
			switch (dia) {
				case 0: aux="domingo"; break;
				case 1: aux="lunes"; break;
				case 2: aux="martes"; break;
				case 3: aux="miércoles"; break;
				case 4: aux="jueves"; break;
				case 5: aux="viernes"; break;
				case 6: aux="sábado"; break;
				default: aux="Error";
			}
			return aux;
		}
		
		function Dia_mes(mes){
			// 0=Enero y 11=diciembre
			var aux="";
			switch (mes) {
				case 0: aux="Enero"; break;
				case 1: aux="Febrero"; break;
				case 2: aux="Marzo"; break;
				case 3: aux="Abril"; break;
				case 4: aux="Mayo"; break;
				case 5: aux="Junio"; break;
				case 6: aux="Julio"; break;
				case 7: aux="Agosto"; break;
				case 8: aux="Septiembre"; break;
				case 9: aux="Octubre"; break;
				case 10: aux="Noviembre"; break;
				case 11: aux="Diciembre"; break;
				default: aux="Error";
			}
			return aux;
		}
		
		function NuevaFecha(form){
			var diaMes = form.diames.value;
			var Mes = form.mes.value;
			var Year = form.year.value;
			var fechaAux = new Date(Year, Mes, diaMes);			
			var segundos = form.seg.value;
			fechaAux.setSeconds(segundos);
			var hora = form.hora.value;
			fechaAux.setHours(hora);
			var minutos = form.min.value;
			fechaAux.setMinutes(minutos);
			var ristra = "La fecha introducida es :\n";
			ristra += Dia_semana(fechaAux.getDay()) + ", " ;
			ristra += fechaAux.getDate() + " de " ;
			ristra += Dia_mes(fechaAux.getMonth()) + " del año " 
			ristra += fechaAux.getYear() + "\n";
			ristra += fechaAux.getHours() + ":" + fechaAux.getMinutes() + ":" ;
			ristra += fechaAux.getSeconds();
			alert(ristra);
			return fechaAux;
		}
					
		var fecha1;
		
		function ComparaFecha(fechaActual, fecha1) {
			var msegActual = fechaActual.getTime();
			var msegFecha1 = fecha1.getTime();
			var Diferencia = msegActual - msegFecha1;
			Diferencia /= 86400000;
			if (Diferencia < 0) {
				Diferencia = Math.abs(Diferencia);
				alert("La fecha introducida es " + Diferencia + 
					" días mayor \n que la fecha actual.");
			}
			else {
				if (Diferencia !=0) 
					alert("La fecha actual es " + Diferencia + 
						" días mayor \n que la introducida.");
				else 
					alert("Las fechas son iguales.");
			}
		}
		
		function Convierte(form, dia) {
			var ristra = dia.toString();
			form.conv.value = ristra;
		}

		function ConvierteGMT(form, dia) {
			var ristra = dia.toGMTString();
			form.conv.value = ristra;
		}

		function ConvierteLocal(form, dia) {
			var ristra = dia.toLocaleString();
			form.conv.value = ristra;
		}
		
		function Conv_fecha(form) {
			NewFecha = Date.parse(form.conv.value);
			var ristra = "El número de milisegundos es de:\n";
			ristra += NewFecha;
			alert(ristra);			
			return NewFecha;
		}
		
		function Cal_Dif_GMT(fechaLocal){
			var diferencia = fechaLocal.getTimezoneOffset();
			var ristra = "La diferencia es:\n";
			ristra += diferencia;
			alert(ristra);
			return diferencia;
		}
		
		function CalculaMili(form){
			var year = form.year.value;
			var mes = form.mes.value;
			var hora = form.hora.value;
			var min = form.min.value;
			var seg = form.seg.value;
			var diaMes = form.diames.value;
			var milisegundos = Date.UTC(year,mes,diaMes,hora,min,seg);
			var ristra = "Los milisegundos transcurridos son:\n";
			ristra += milisegundos;
			alert(ristra);
			return milisegundos;
		}
